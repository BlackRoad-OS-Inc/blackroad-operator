import { useState, useEffect, useRef } from 'react';

const EMOJIS = ['◈', '⌗', '⧖', '⫯', '⩮', '⫱', '⧓', '⧗', '⧕', '∞', '◉', '❋'];
const COLORS = ['#ff8700', '#ff5f00', '#ff0087', '#d700af', '#0087ff', '#ffffff'];

export default function SymmetryOS() {
  const [nodes, setNodes] = useState([]);
  const [rotation, setRotation] = useState({ x: 0, y: 0 });
  const [ollamaState, setOllamaState] = useState('awaiting');
  const [pulsePhase, setPulsePhase] = useState(0);
  const containerRef = useRef(null);

  // Initialize symmetric nodes
  useEffect(() => {
    const newNodes = [];
    const layers = 3;
    const nodesPerLayer = 6;
    
    for (let layer = 0; layer < layers; layer++) {
      for (let i = 0; i < nodesPerLayer; i++) {
        const angle = (i / nodesPerLayer) * Math.PI * 2;
        const radius = 120 + layer * 60;
        const z = (layer - 1) * 80;
        
        newNodes.push({
          id: `${layer}-${i}`,
          x: Math.cos(angle) * radius,
          y: Math.sin(angle) * radius,
          z: z,
          emoji: EMOJIS[Math.floor(Math.random() * EMOJIS.length)],
          color: COLORS[layer % COLORS.length],
          phase: i / nodesPerLayer,
          layer
        });
        
        // Mirror node for perfect symmetry
        newNodes.push({
          id: `${layer}-${i}-mirror`,
          x: Math.cos(angle + Math.PI) * radius,
          y: Math.sin(angle + Math.PI) * radius,
          z: -z,
          emoji: EMOJIS[Math.floor(Math.random() * EMOJIS.length)],
          color: COLORS[(layer + 1) % COLORS.length],
          phase: (i + nodesPerLayer/2) / nodesPerLayer,
          layer,
          mirror: true
        });
      }
    }
    setNodes(newNodes);
  }, []);

  // Animation loop
  useEffect(() => {
    let frame;
    const animate = () => {
      setRotation(r => ({
        x: r.x + 0.003,
        y: r.y + 0.007
      }));
      setPulsePhase(p => (p + 0.02) % (Math.PI * 2));
      frame = requestAnimationFrame(animate);
    };
    frame = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(frame);
  }, []);

  // Simulated Ollama control
  useEffect(() => {
    const interval = setInterval(() => {
      setNodes(prev => prev.map(node => ({
        ...node,
        emoji: EMOJIS[Math.floor(Math.random() * EMOJIS.length)]
      })));
      setOllamaState(prev => 
        prev === 'awaiting' ? 'processing' : 
        prev === 'processing' ? 'emitting' : 'awaiting'
      );
    }, 2000);
    return () => clearInterval(interval);
  }, []);

  const project = (node) => {
    const cosX = Math.cos(rotation.x);
    const sinX = Math.sin(rotation.x);
    const cosY = Math.cos(rotation.y);
    const sinY = Math.sin(rotation.y);
    
    let { x, y, z } = node;
    
    // Rotate around Y
    const x1 = x * cosY - z * sinY;
    const z1 = x * sinY + z * cosY;
    
    // Rotate around X
    const y1 = y * cosX - z1 * sinX;
    const z2 = y * sinX + z1 * cosX;
    
    const scale = 400 / (400 + z2);
    const pulse = Math.sin(pulsePhase + node.phase * Math.PI * 2) * 0.3 + 1;
    
    return {
      x: x1 * scale,
      y: y1 * scale,
      scale: scale * pulse,
      z: z2,
      opacity: 0.4 + scale * 0.6
    };
  };

  const sortedNodes = [...nodes].sort((a, b) => {
    const pa = project(a);
    const pb = project(b);
    return pa.z - pb.z;
  });

  return (
    <div style={{
      width: '100%',
      height: '100vh',
      background: 'linear-gradient(135deg, #0a0a0a 0%, #1a0a1a 50%, #0a0a1a 100%)',
      display: 'flex',
      flexDirection: 'column',
      alignItems: 'center',
      justifyContent: 'center',
      fontFamily: 'monospace',
      overflow: 'hidden'
    }}>
      {/* Header */}
      <div style={{
        position: 'absolute',
        top: 20,
        color: '#ff8700',
        fontSize: 14,
        textAlign: 'center'
      }}>
        <div style={{ letterSpacing: 4 }}>⌗ BLACKROAD OS</div>
        <div style={{ color: '#666', fontSize: 11, marginTop: 4 }}>
          PERFECT SYMMETRY • PS-SHA∞
        </div>
      </div>

      {/* Ollama Status */}
      <div style={{
        position: 'absolute',
        top: 80,
        display: 'flex',
        gap: 20,
        alignItems: 'center'
      }}>
        <div style={{
          padding: '8px 16px',
          background: ollamaState === 'emitting' ? '#ff870022' : '#ffffff08',
          border: `1px solid ${ollamaState === 'emitting' ? '#ff8700' : '#333'}`,
          borderRadius: 4,
          color: ollamaState === 'emitting' ? '#ff8700' : '#666',
          fontSize: 11,
          transition: 'all 0.3s'
        }}>
          OLLAMA: {ollamaState.toUpperCase()}
        </div>
        <div style={{
          width: 8,
          height: 8,
          borderRadius: '50%',
          background: ollamaState === 'emitting' ? '#ff8700' : 
                      ollamaState === 'processing' ? '#d700af' : '#333',
          boxShadow: ollamaState === 'emitting' ? '0 0 20px #ff8700' : 'none',
          transition: 'all 0.3s'
        }} />
      </div>

      {/* 3D Container */}
      <div 
        ref={containerRef}
        style={{
          position: 'relative',
          width: 500,
          height: 500,
          perspective: 1000
        }}
      >
        {/* Center core */}
        <div style={{
          position: 'absolute',
          left: '50%',
          top: '50%',
          transform: 'translate(-50%, -50%)',
          width: 60,
          height: 60,
          borderRadius: '50%',
          background: 'radial-gradient(circle, #ff8700 0%, #d700af 50%, transparent 70%)',
          opacity: 0.6 + Math.sin(pulsePhase) * 0.4,
          boxShadow: '0 0 60px #ff870066',
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          fontSize: 24,
          color: '#fff'
        }}>
          ◈
        </div>

        {/* Orbital rings */}
        {[1, 2, 3].map(ring => (
          <div key={ring} style={{
            position: 'absolute',
            left: '50%',
            top: '50%',
            width: ring * 120,
            height: ring * 120,
            border: `1px solid ${COLORS[ring - 1]}22`,
            borderRadius: '50%',
            transform: `translate(-50%, -50%) rotateX(${rotation.x * 30}deg) rotateY(${rotation.y * 30}deg)`,
            transition: 'transform 0.1s linear'
          }} />
        ))}

        {/* Nodes */}
        {sortedNodes.map(node => {
          const p = project(node);
          return (
            <div
              key={node.id}
              style={{
                position: 'absolute',
                left: '50%',
                top: '50%',
                transform: `translate(-50%, -50%) translate(${p.x}px, ${p.y}px) scale(${p.scale})`,
                fontSize: 24,
                color: node.color,
                opacity: p.opacity,
                textShadow: `0 0 20px ${node.color}`,
                transition: 'transform 0.05s linear',
                cursor: 'pointer',
                filter: node.mirror ? 'hue-rotate(30deg)' : 'none'
              }}
            >
              {node.emoji}
            </div>
          );
        })}

        {/* Connection lines */}
        <svg style={{
          position: 'absolute',
          top: 0,
          left: 0,
          width: '100%',
          height: '100%',
          pointerEvents: 'none'
        }}>
          {nodes.slice(0, 12).map((node, i) => {
            const p1 = project(node);
            const nextNode = nodes[(i + 2) % 12];
            if (!nextNode) return null;
            const p2 = project(nextNode);
            return (
              <line
                key={`line-${i}`}
                x1={250 + p1.x}
                y1={250 + p1.y}
                x2={250 + p2.x}
                y2={250 + p2.y}
                stroke={node.color}
                strokeWidth={0.5}
                opacity={0.2}
              />
            );
          })}
        </svg>
      </div>

      {/* Footer controls */}
      <div style={{
        position: 'absolute',
        bottom: 30,
        display: 'flex',
        gap: 30,
        color: '#666',
        fontSize: 11
      }}>
        {['⊳ EMIT', '⊲ ABSORB', '⧗ BRANCH', '∞ RECURSE'].map((cmd, i) => (
          <div key={cmd} style={{
            padding: '6px 12px',
            border: '1px solid #333',
            borderRadius: 3,
            cursor: 'pointer',
            color: COLORS[i],
            transition: 'all 0.2s'
          }}>
            {cmd}
          </div>
        ))}
      </div>
    </div>
  );
}
