def calculate_average(numbers):
    """
    Calculate and return the average of a list of numbers.
    
    :param numbers: A list of numerical values.
    :return: The average value as a float.
    """
    if not numbers:
        raise ValueError("The list of numbers cannot be empty.")
    total = sum(numbers)
    count = len(numbers)
    return total / count
