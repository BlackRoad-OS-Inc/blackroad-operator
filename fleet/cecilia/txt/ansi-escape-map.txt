#!/usr/bin/env bash
# ANSI Escape Code Mapper & Tester
# Maps all escape sequences and tests terminal compatibility

# ==================
# ESCAPE CODE MAP
# ==================

cat << 'EOF'
╔════════════════════════════════════════════════════════════════╗
║          ANSI ESCAPE SEQUENCE REFERENCE MAP                    ║
╚════════════════════════════════════════════════════════════════╝

[CONTROL SEQUENCES]
  Clear screen:     \033[2J       or ESC[2J
  Home cursor:      \033[H        or ESC[H
  Clear & home:     \033[2J\033[H or ESC[2JESC[H
  
  Move cursor:      \033[<row>;<col>H
  Move up:          \033[<n>A
  Move down:        \033[<n>B
  Move right:       \033[<n>C
  Move left:        \033[<n>D
  
  Clear to end:     \033[K
  Save cursor:      \033[s
  Restore cursor:   \033[u
  Hide cursor:      \033[?25l
  Show cursor:      \033[?25h

[TEXT ATTRIBUTES]
  Reset:            \033[0m
  Bold:             \033[1m
  Dim:              \033[2m
  Italic:           \033[3m
  Underline:        \033[4m
  Blink:            \033[5m
  Reverse:          \033[7m
  Hidden:           \033[8m
  Strikethrough:    \033[9m

[BASIC COLORS (3/4 bit)]
  Format:           \033[<code>m

  Foreground (30-37):
    30 = Black      34 = Blue
    31 = Red        35 = Magenta
    32 = Green      36 = Cyan
    33 = Yellow     37 = White
  
  Background (40-47):
    40 = Black      44 = Blue
    41 = Red        45 = Magenta
    42 = Green      46 = Cyan
    43 = Yellow     47 = White

  Bright Foreground (90-97):
    90 = Bright Black    94 = Bright Blue
    91 = Bright Red      95 = Bright Magenta
    92 = Bright Green    96 = Bright Cyan
    93 = Bright Yellow   97 = Bright White

[256 COLOR MODE]
  Foreground:       \033[38;5;<n>m
  Background:       \033[48;5;<n>m
  
  Where <n> is:
    0-15:    Standard colors
    16-231:  216 color cube (6×6×6)
    232-255: Grayscale

[TRUE COLOR (24-bit RGB)]
  Foreground:       \033[38;2;<r>;<g>;<b>m
  Background:       \033[48;2;<r>;<g>;<b>m
  
  Where r,g,b are 0-255

[BLACKROAD COLORS - 256 MODE]
  Pink:             \033[38;5;205m   (#FF1D6C)
  Amber:            \033[38;5;214m   (#F5A623)
  Blue:             \033[38;5;75m    (#2979FF)
  Violet:           \033[38;5;141m   (#9C27B0)
  Purple:           \033[38;5;141m   (same as violet)
  Orange:           \033[38;5;208m   (#FF6B35)
  Gray:             \033[38;5;240m   
  Light Gray:       \033[38;5;250m   
  White:            \033[38;5;255m   

[BLACKROAD COLORS - TRUE COLOR]
  Pink:             \033[38;2;255;29;108m
  Amber:            \033[38;2;245;166;35m
  Blue:             \033[38;2;41;121;255m
  Violet:           \033[38;2;156;39;176m
  Orange:           \033[38;2;255;107;53m

[USAGE IN BASH]
  # Method 1: echo -e
  echo -e "\033[31mRed text\033[0m"
  
  # Method 2: printf
  printf "\033[31mRed text\033[0m\n"
  
  # Method 3: $'...' syntax
  echo $'\033[31mRed text\033[0m'
  
  # Method 4: Variables
  RED='\033[31m'
  RESET='\033[0m'
  echo -e "${RED}Red text${RESET}"

[COMMON MISTAKES]
  ❌ echo "\033[31m"        # Missing -e flag
  ❌ printf "${RED}"        # Using echo variable in printf
  ❌ \e[31m                 # \e not always supported, use \033
  ✅ echo -e "\033[31m"
  ✅ printf "\033[31m"
  ✅ echo $'\033[31m'

[TERMINAL COMPATIBILITY]
  Check support:
    echo $TERM
    tput colors
  
  Safe fallback:
    if [[ -t 1 ]] && [[ $(tput colors 2>/dev/null) -ge 8 ]]; then
      # Use colors
    fi

[BOX DRAWING CHARACTERS]
  Requires UTF-8 terminal
  
  Single line:
    ┌ ─ ┐
    │   │
    └ ─ ┘
  
  Double line:
    ╔ ═ ╗
    ║   ║
    ╚ ═ ╝
  
  Mixed:
    ╔═╦═╗
    ║ ║ ║
    ╠═╬═╣
    ║ ║ ║
    ╚═╩═╝
  
  Codes:
    ╔ = U+2554    ═ = U+2550    ╗ = U+2557
    ║ = U+2551                  
    ╚ = U+255A    ═ = U+2550    ╝ = U+255D
    ╠ = U+2560    ╬ = U+256C    ╣ = U+2563
    ╦ = U+2566    ╩ = U+2569

EOF
