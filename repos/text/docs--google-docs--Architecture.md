#include

#include

#define LED_PIN 6

#define LED_COUNT 3

#define COIL_PIN 9

#define SENSOR_PIN A0

#define RX_PIN 10

#define TX_PIN 11

Adafruit_NeoPixel strip(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800);

SoftwareSerial piSerial(RX_PIN, TX_PIN);

void setup() {

strip.begin();

strip.show();

pinMode(COIL_PIN, OUTPUT);

pinMode(SENSOR_PIN, INPUT);

piSerial.begin(9600);

}

void loop() {

int sensorValue = analogRead(SENSOR_PIN);

float normalized = sensorValue / 1023.0;

piSerial.print(normalized, 2);

piSerial.println();

if (piSerial.available()) {

String cmd = piSerial.readStringUntil('\n');

if (cmd.startsWith("LED")) {

int r = cmd.substring(4, 7).toInt();

int g = cmd.substring(8, 11).toInt();

int b = cmd.substring(12, 15).toInt();

for (int i = 0; i < LED_COUNT; i++) {

strip.setPixelColor(i, strip.Color(r, g, b));

}

strip.show();

} else if (cmd.startsWith("COIL")) {

int duty = cmd.substring(5).toInt();

analogWrite(COIL_PIN, duty);

}

}

delay(10);

}
