... import numpy as np

... def sigmoid(z):

...     return 1 / (1 + np.exp(-z))

... class LogisticRegressionGD:

...     def __init__(self, lr=0.1, epochs=1000, reg=0.0):

...         self.lr = lr

...         self.epochs = epochs

...         self.reg = reg  # L2 lambda

...

...     def fit(self, X, y):

...         n, d = X.shape

...         self.w = np.zeros(d)

...         self.b = 0.0

...         for _ in range(self.epochs):

...             # forward

...             z = X @ self.w + self.b

...             p = sigmoid(z)

...             # gradients (binary cross-entropy + L2 on w)

...             grad_w = (X.T @ (p - y)) / n + self.reg * self.w

...             grad_b = np.mean(p - y)

...             # update

...             self.w -= self.lr * grad_w

...             self.b -= self.lr * grad_b

...         return self

...

...     def predict_proba(self, X):

...         return sigmoid(X @ self.w + self.b)

...

...     def predict(self, X, thresh=0.5):

...         return (self.predict_proba(X) >= thresh).astype(int)
